完整数据库架构设计

包含用户、小说内容、RAG、知识库的完整数据模型

📋 数据分类
结构化数据（数据库）：用户账号、项目元数据、章节列表、角色/世界观索引
大文本内容（文件系统）：章节正文、Story Bible 完整文档、素材库原文
向量数据（专用存储）：RAG embeddings、语义检索索引
-- ============================================
-- 用户与权限模块
-- ============================================
CREATE TABLE
users (
id TEXT PRIMARY KEY,
username TEXT UNIQUE NOT NULL,
email TEXT UNIQUE NOT NULL,
password_hash TEXT NOT NULL,
role TEXT NOT NULL,
storage_quota_mb INTEGER DEFAULT 5120, -- 5GB
api_call_quota INTEGER DEFAULT 10000,
created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
is_active BOOLEAN DEFAULT 1
);
CREATE TABLE
user_settings (
user_id TEXT PRIMARY KEY,
-- LLM配置
llm_provider TEXT, llm_api_key_encrypted TEXT, llm_model TEXT,
-- Embedding配置
embedding_provider TEXT DEFAULT 'local', embedding_model TEXT,
-- 其他配置...
FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);
-- ============================================
-- 小说项目模块
-- ============================================
CREATE TABLE
novel_projects (
id TEXT PRIMARY KEY,
user_id TEXT NOT NULL,
title TEXT NOT NULL,
genre TEXT,
sub_genre TEXT,
core_hook TEXT,
status TEXT DEFAULT 'draft', -- draft/writing/completed
word_count INTEGER DEFAULT 0,
cover_image_url TEXT,
created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);
CREATE TABLE
volumes (
id TEXT PRIMARY KEY,
project_id TEXT NOT NULL,
title TEXT NOT NULL,
volume_number INTEGER NOT NULL,
summary TEXT,
FOREIGN KEY (project_id) REFERENCES novel_projects(id) ON DELETE CASCADE
);
CREATE TABLE
chapters (
id TEXT PRIMARY KEY,
project_id TEXT NOT NULL,
volume_id TEXT,
title TEXT NOT NULL,
chapter_number INTEGER NOT NULL,
word_count INTEGER DEFAULT 0,
status TEXT DEFAULT 'draft', -- draft/writing/completed
content_path TEXT NOT NULL, -- 文件系统路径
created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (project_id) REFERENCES novel_projects(id) ON DELETE CASCADE
);
-- ============================================
-- Story Bible 模块（角色、世界观）
-- ============================================
CREATE TABLE
characters (
id TEXT PRIMARY KEY,
project_id TEXT NOT NULL,
name TEXT NOT NULL,
role TEXT, -- protagonist/antagonist/supporting
description TEXT,
personality TEXT,
background TEXT,
relationships TEXT, -- JSON格式
first_appearance_chapter TEXT,
FOREIGN KEY (project_id) REFERENCES novel_projects(id) ON DELETE CASCADE
);
CREATE TABLE
world_settings (
id TEXT PRIMARY KEY,
project_id TEXT NOT NULL,
category TEXT NOT NULL, -- location/power_system/culture/etc
name TEXT NOT NULL,
description TEXT,
rules TEXT, -- JSON格式
FOREIGN KEY (project_id) REFERENCES novel_projects(id) ON DELETE CASCADE
);
CREATE TABLE
plot_threads (
id TEXT PRIMARY KEY,
project_id TEXT NOT NULL,
name TEXT NOT NULL,
description TEXT,
setup_chapter TEXT,
payoff_chapter TEXT,
status TEXT DEFAULT 'setup', -- setup/developing/resolved
FOREIGN KEY (project_id) REFERENCES novel_projects(id) ON DELETE CASCADE
);
-- ============================================
-- 素材库与知识库模块
-- ============================================
CREATE TABLE
materials (
id TEXT PRIMARY KEY,
user_id TEXT NOT NULL,
project_id TEXT, -- NULL表示全局素材
title TEXT NOT NULL,
category TEXT, -- 参考资料/灵感/设定/etc
content_type TEXT, -- text/image/url
content TEXT, -- 短文本直接存，长文本存路径
file_path TEXT,
tags TEXT, -- JSON数组
created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);
CREATE TABLE
knowledge_base (
id TEXT PRIMARY KEY,
project_id TEXT NOT NULL,
entry_type TEXT NOT NULL, -- 写作技巧/情节模板/对话范例
title TEXT NOT NULL,
content TEXT NOT NULL,
usage_count INTEGER DEFAULT 0,
created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (project_id) REFERENCES novel_projects(id) ON DELETE CASCADE
);
-- ============================================
-- RAG 向量检索模块
-- ============================================
CREATE TABLE
rag_chunks (
id TEXT PRIMARY KEY,
project_id TEXT NOT NULL,
source_type TEXT NOT NULL, -- chapter/character/world/material
source_id TEXT NOT NULL, -- 关联的章节/角色/世界观ID
chunk_index INTEGER NOT NULL,
content TEXT NOT NULL, -- 分块后的文本
embedding_vector BLOB, -- 向量数据（或存向量数据库）
created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (project_id) REFERENCES novel_projects(id) ON DELETE CASCADE
);
-- 为 RAG 检索创建索引
CREATE INDEX
idx_rag_source ON rag_chunks(source_type, source_id);
CREATE INDEX
idx_rag_project ON rag_chunks(project_id);
-- ============================================
-- Agent 记忆模块
-- ============================================
CREATE TABLE
agent_memories (
id TEXT PRIMARY KEY,
user_id TEXT NOT NULL,
project_id TEXT, -- NULL表示author_memory（全局）
memory_type TEXT NOT NULL, -- author/project/execution
content TEXT NOT NULL, -- JSON格式
updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);
CREATE TABLE
agent_sessions (
id TEXT PRIMARY KEY,
user_id TEXT NOT NULL,
project_id TEXT,
session_data TEXT, -- JSON格式
created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

💾 文件系统组织

App_Data/

│ └── novelagent.db ← SQLite数据库文件
└── Users/
└── [userId]/
├── Projects/
│ └── [projectId]/
│ ├── chapters/ ← 章节正文（.md文件）
│ ├── storybible/ ← Story Bible完整JSON
│ └── materials/ ← 素材原文件
